cmake_minimum_required(VERSION 3.16)
project(aurora_watch_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_SKIP_RPATH TRUE)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick)
add_executable(aurora-preview main.cpp)
target_link_libraries(aurora-preview PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/../../stage27/qml" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
configure_file(Main.qml "${CMAKE_CURRENT_BINARY_DIR}/qml/Main.qml" COPYONLY)
include(CTest)
if(BUILD_TESTING)
    add_executable(aurora-session-tests session_tests.cpp)
    target_link_libraries(aurora-session-tests PRIVATE Qt6::Core)
    if(NOT CMAKE_CROSSCOMPILING)
        add_test(NAME gui-session-discovery COMMAND aurora-session-tests)
    endif()
endif()
