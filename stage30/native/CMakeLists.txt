cmake_minimum_required(VERSION 3.16)
project(aurora_shell LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_SKIP_RPATH TRUE)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick DBus Network Sensors)
add_library(aurorasystemplugin SHARED plugin.cpp system.cpp system.hpp)
target_link_libraries(aurorasystemplugin PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::DBus Qt6::Network Qt6::Sensors)
set_target_properties(aurorasystemplugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/qml/Aurora/System")
add_executable(aurora-helper helper.c)
set_target_properties(aurora-helper PROPERTIES AUTOMOC OFF)
add_executable(aurora-preview preview.cpp)
target_link_libraries(aurora-preview PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
add_executable(aurora-state-tests state_tests.cpp)
target_link_libraries(aurora-state-tests PRIVATE Qt6::Core)
add_executable(aurora-session-tests ../../stage28/native/session_tests.cpp)
target_link_libraries(aurora-session-tests PRIVATE Qt6::Core)
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/../qml" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qml/Aurora/System/qmldir" "module Aurora.System\nplugin aurorasystemplugin\n")
