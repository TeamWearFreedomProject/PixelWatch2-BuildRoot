cmake_minimum_required(VERSION 3.16)
project(aurora_home LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_SKIP_RPATH TRUE)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick)
add_executable(aurora-home main.cpp)
target_link_libraries(aurora-home PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/../qml" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/../tests/Fixture.qml" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/qml")
add_executable(aurora-home-tests home_tests.cpp)
target_link_libraries(aurora-home-tests PRIVATE Qt6::Core)
add_executable(aurora-session-tests ../../stage28/native/session_tests.cpp)
target_link_libraries(aurora-session-tests PRIVATE Qt6::Core)
